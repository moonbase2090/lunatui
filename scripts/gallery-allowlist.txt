# Gallery guard allowlist (see scripts/check-gallery-real-output.sh).
#
# Each line names one thing and says why it is safe.
#   helper NAME # reason   a function that takes a render surface (RenderCx,
#                          Grid, Frame, Buffer) and only lays out real widgets
#   scope PATH # reason    an extra example file the guard must scan
#
# A helper may not write cells itself. The cell-writing calls stay forbidden
# everywhere in scope, so listing a helper never lets it paint.
scope examples/menu_showcase.rs # the editor scene is composed entirely from public widgets
helper new # resolves semantic colors for the editor scene without writing cells
helper panel # lays out and renders a Block
helper draw # initializes a frame and composes the editor, Menu and status widgets
helper explorer # renders the project List and text labels
helper editor # renders code Paragraphs, Tabs and a TextInput caret
helper activity # renders application event messages through Paragraph
scope examples/kitty_image.rs # the native pixel capture composes Image, Block and Paragraph widgets
scope examples/pixel_chart.rs # live native Chart demo uses the runtime image presenter
scope examples/chart_scene.rs # shared chart scene uses Chart, Block and typed datasets
helper throughput # constructs Chart request line datasets and renders the widget
helper saturation # constructs Chart CPU/memory datasets and renders the widget
helper scatter # constructs Chart latency datasets and renders the widget
helper histogram # constructs Chart hourly datasets and renders the widget
scope examples/dialog_showcase.rs # the branch manager scene composes Block, Paragraph, LineGauge, StatusBar and Dialog widgets

helper clock_and_tree # draws clock and tree shapes through Canvas
